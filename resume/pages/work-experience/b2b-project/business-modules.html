方案与实现

1. 合同全生命周期与状态机

合同不是一张表，而是一台状态机：每个动作（提交/审批/签章/付款/配货/发货/收货/开票/完成/取消）都只能从合法前置态迁移。我们把状态迁移规则集中到一处校验，避免几十个 if-else 散落在各服务里——这是后续接 Saga 长流程的地基。 

/**
 * 合同状态机：集中定义合法迁移，所有业务动作统一走 transition()
 * 我在合同中心落地的第一件事，就是把散落的 if 判断收拢成一张迁移表。
 */
@Component
public class ContractStateMachine {

    // 前置态 -> 允许的后继态
    private static final Map<ContractStatus, Set<ContractStatus>> TRANSITIONS = Map.of(
        ContractStatus.DRAFT,     Set.of(ContractStatus.PENDING, ContractStatus.CANCELLED),
        ContractStatus.PENDING,   Set.of(ContractStatus.APPROVED, ContractStatus.REJECTED),
        ContractStatus.APPROVED,  Set.of(ContractStatus.SIGNED, ContractStatus.CANCELLED),
        ContractStatus.SIGNED,    Set.of(ContractStatus.PAID, ContractStatus.SHIPPED),
        ContractStatus.PAID,      Set.of(ContractStatus.ALLOCATED, ContractStatus.SHIPPED),
        ContractStatus.ALLOCATED, Set.of(ContractStatus.SHIPPED),
        ContractStatus.SHIPPED,   Set.of(ContractStatus.DELIVERED),
        ContractStatus.DELIVERED, Set.of(ContractStatus.INVOICED, ContractStatus.COMPLETED),
        ContractStatus.INVOICED,  Set.of(ContractStatus.COMPLETED)
    );

    public void transition(Contract contract, ContractStatus target) {
        ContractStatus from = contract.getStatus();
        if (!TRANSITIONS.getOrDefault(from, Set.of()).contains(target)) {
            throw new BusinessException(
                "非法状态迁移: " + from.getDesc() + " -> " + target.getDesc());
        }
        contract.setStatus(target);
    }
}

public enum ContractStatus {
    DRAFT("草稿"), PENDING("待审批"), APPROVED("已审批"), REJECTED("已拒绝"),
    SIGNED("已签章"), PAID("已付款"), ALLOCATED("已配货"), SHIPPED("已发货"),
    DELIVERED("已收货"), INVOICED("已开票"), COMPLETED("已完成"), CANCELLED("已取消");
    private final String desc;
    ContractStatus(String desc) { this.desc = desc; }
    public String getDesc() { return desc; }
}

public enum SettlementType {
    ORGANIZED("组织交收"), FREE("自由交收");
    private final String desc;
    SettlementType(String desc) { this.desc = desc; }
    public String getDesc() { return desc; }
}

2. 两种交收方式：组织交收 vs 自由交收

交收方式由客户配置决定，代码里只保留一个策略分派点：组织交收走"付款→仓单过户→交收完成"，自由交收走"签章→自行发货→物流→确认收货"。同一套状态机，两种履约路径，客户切换交收方式无需改代码。

组织交收

审批 签章 支付 仓单过户 完成

自由交收

审批 签章 自行发货 物流 收货 完成 
@Service
public class SettlementService {

    @Autowired private ContractStateMachine stateMachine;
    @Autowired private WarehouseReceiptService warehouseReceiptService;
    @Autowired private LogisticsService logisticsService;

    /** 交收方式来自租户配置，这里只做策略分派 */
    public void start(SettlementContext ctx) {
        Contract c = ctx.getContract();
        if (c.getSettlementType() == SettlementType.ORGANIZED) {
            // 组织交收：付款后按仓单过户推进
            stateMachine.transition(c, ContractStatus.SIGNED);
        } else {
            // 自由交收：签章后走自行发货
            stateMachine.transition(c, ContractStatus.SIGNED);
        }
    }

    /** 组织交收：仓单批量过户到买方，一次 Saga 事务 */
    public void allocateWarehouseReceipt(Contract c, List<String> receiptIds) {
        stateMachine.transition(c, ContractStatus.ALLOCATED); // 校验必须为已付款
        for (String id : receiptIds) {
            warehouseReceiptService.transfer(id, c.getBuyerId());
        }
    }

    /** 自由交收：创建物流单 */
    public String arrangeDelivery(Contract c, DeliveryRequest req) {
        stateMachine.transition(c, ContractStatus.SHIPPED); // 校验必须为已签章
        return logisticsService.createOrder(req.getLogisticsCompany(),
                req.getSenderAddress(), req.getReceiverAddress(), req.getItems());
    }
}

3. 电子签章（对接外部签章系统）

签章系统属于外部集成（外联网关统一适配），合同中心只负责双签状态编排：买卖双方各自签署完成后，合同才进入 SIGNED。签章回调通过外联网关统一入口、幂等消费，避免重复签章导致状态错乱。 

@Service
public class ContractSignatureService {

    @Autowired private ContractStateMachine stateMachine;
    @Autowired private ExternalGateway externalGateway; // 外联网关统一适配签章系统

    public SignResult sign(String contractId, Long signerId, SignerRole role) {
        Contract c = contractService.getContract(contractId);
        if (c.getStatus() != ContractStatus.APPROVED) {
            throw new BusinessException("只有已审批的合同可以签章");
        }

        // 经网关调用签章系统，返回签署链接
        ExternalResponse resp = externalGateway.invoke("esign", "sign",
                Map.of("contractId", contractId, "signerId", signerId, "role", role));
        if (!resp.isSuccess()) {
            return SignResult.fail(resp.getMessage());
        }

        if (role == SignerRole.BUYER) {
            c.setBuyerSigned(true);  c.setBuyerSignTime(new Date());
        } else {
            c.setSellerSigned(true); c.setSellerSignTime(new Date());
        }
        // 双签完成才推进状态机
        if (c.isBuyerSigned() && c.isSellerSigned()) {
            stateMachine.transition(c, ContractStatus.SIGNED);
            c.setSignTime(new Date());
        }
        contractService.update(c);
        return SignResult.ok(resp.getData());
    }
}

public enum SignerRole { BUYER, SELLER }

4. 合同配货：仓单过户 + 物流对接

配货要保证合同标的多数量 = 实际分配的仓单数量，数量不匹配直接拦截；配货成功后创建物流单并回传轨迹。物流 WMS 与仓单系统都在外联网关适配层之下，换供应商不影响合同中心。 

@Service
public class ContractAllocationService {

    @Autowired private ContractStateMachine stateMachine;
    @Autowired private ContractItemRepository contractItemRepository;

    public AllocationResult allocateGoods(String contractId, AllocationRequest req) {
        Contract c = contractService.getContract(contractId);
        if (c.getStatus() != ContractStatus.PAID) {
            throw new BusinessException("只有已付款的合同可以配货");
        }

        List<ContractItem> items = contractItemRepository.findByContractId(contractId);
        for (ContractItem item : items) {
            AllocationRequest.Item r = req.findByProduct(item.getProductId());
            if (r == null) {
                throw new BusinessException("商品未分配: " + item.getProductName());
            }
            // 数量必须严格匹配，避免"少配多配"埋下资金差
            if (r.getQuantity().compareTo(item.getQuantity()) != 0) {
                throw new BusinessException("数量不匹配: " + item.getProductName());
            }
            warehouseReceiptService.transfer(r.getWarehouseReceiptId(), c.getBuyerId());
        }

        stateMachine.transition(c, ContractStatus.ALLOCATED);
        c.setAllocationTime(new Date());
        contractService.update(c);
        return AllocationResult.ok();
    }
}

5. 税务开票（对接开票服务）

收货/完成后才允许申请开票，发票申请单落本地库、经外联网关提交税务开票平台，异步回调更新发票号与状态。开票金额支持按合同全额或按短溢后实际金额，与资金模块共用同一套计算口径。 

@Service
public class ContractInvoiceService {

    @Autowired private ContractStateMachine stateMachine;
    @Autowired private ExternalGateway externalGateway; // 对接税务开票平台

    public InvoiceResult applyInvoice(String contractId, InvoiceRequest req) {
        Contract c = contractService.getContract(contractId);
        if (c.getStatus() != ContractStatus.DELIVERED
                && c.getStatus() != ContractStatus.COMPLETED) {
            throw new BusinessException("只有已收货或已完成的合同可以开票");
        }

        InvoiceApplication app = InvoiceApplication.builder()
            .contractId(contractId)
            .invoiceType(req.getInvoiceType())
            .invoiceAmount(resolveInvoiceAmount(c, req)) // 全额 or 短溢后实际额
            .status(InvoiceApplicationStatus.PENDING)
            .createTime(new Date())
            .build();
        invoiceApplicationRepository.save(app);

        ExternalResponse resp = externalGateway.invoke("tax-invoice", "apply", app.toMap());
        if (resp.isSuccess()) {
            app.setInvoiceNo((String) resp.getData().get("invoiceNo"));
            app.setStatus(InvoiceApplicationStatus.ISSUED);
            stateMachine.transition(c, ContractStatus.INVOICED);
            c.setInvoiceTime(new Date());
        }
        return InvoiceResult.of(resp, app);
    }
}

public enum InvoiceApplicationStatus {
    PENDING("待申请"), APPLIED("已申请"), ISSUED("已开具"), FAILED("申请失败");
    private final String desc;
    InvoiceApplicationStatus(String desc) { this.desc = desc; }
}

6. 复杂资金组合方式

一张合同可能同时涉及预付款、手续费、货款、短溢货款、开票保证金五种资金。我的做法是：资金规则全部配置化（预付比例、费率、保证金比例按租户/合同类型配置），每笔资金变动落资金流水，短溢、冻结/解冻都生成不可逆流水，为日终对账提供唯一事实。

资金类型 说明 计算方式
预付款 签约前/后支付的定金 合同金额 × 预付比例（配置）
手续费 平台交易服务费 合同金额 × 手续费率（配置）
货款 货物实际金额 实际发货数量 × 单价
短溢货款 多发/少发产生的差额，需单独对账 （实际数量 − 合同数量）× 单价
开票保证金 开票前冻结、开票后解冻 合同金额 × 保证金比例（配置） 
@Service
public class CapitalService {

    @Autowired private CapitalFlowRepository capitalFlowRepository;

    /** 资金组合计算：所有比例来自租户配置，不写死在代码里 */
    public CapitalCalculationResult calculate(Contract c, List<ContractItem> items) {
        BigDecimal base = c.getTotalAmount();
        BigDecimal prepayment   = base.multiply(c.prepaymentRate());     // 预付款
        BigDecimal serviceFee   = base.multiply(c.serviceFeeRate());     // 手续费
        BigDecimal goodsPayment = items.stream()                         // 货款（按实际数量）
            .map(i -> i.actualOrContractQty().multiply(i.getPrice()))
            .reduce(BigDecimal.ZERO, BigDecimal::add);
        BigDecimal shortOver    = items.stream()                         // 短溢货款
            .map(i -> i.actualOrContractQty().subtract(i.getQuantity()).multiply(i.getPrice()))
            .reduce(BigDecimal.ZERO, BigDecimal::add);
        BigDecimal invoiceDeposit = base.multiply(c.invoiceDepositRate()); // 开票保证金

        return CapitalCalculationResult.builder()
            .contractAmount(base).prepayment(prepayment).serviceFee(serviceFee)
            .goodsPayment(goodsPayment).shortOverPayment(shortOver)
            .invoiceDeposit(invoiceDeposit).build();
    }

    /** 开票保证金冻结/解冻：每次操作都落一条不可逆资金流水，供日终对账 */
    public void freezeInvoiceDeposit(Contract c) {
        CapitalCalculationResult r = calculate(c, itemsOf(c));
        capitalFlowRepository.save(CapitalFlow.of(c.getContractId(),
            "INVOICE_DEPOSIT_FREEZE", r.getInvoiceDeposit(), "开票保证金冻结"));
        c.setInvoiceDepositFrozen(true);
    }

    public void unfreezeInvoiceDeposit(Contract c) {
        if (!c.isInvoiceDepositFrozen()) {
            throw new BusinessException("开票保证金未冻结");
        }
        CapitalCalculationResult r = calculate(c, itemsOf(c));
        capitalFlowRepository.save(CapitalFlow.of(c.getContractId(),
            "INVOICE_DEPOSIT_UNFREEZE", r.getInvoiceDeposit().negate(), "开票保证金解冻"));
        c.setInvoiceDepositFrozen(false);
    }

    /** 短溢货款结算：正数=向买方收，负数=向卖方付 */
    public void settleShortOver(Contract c, BigDecimal settleAmount) {
        String type = settleAmount.signum() >= 0
            ? "SHORT_OVER_RECEIVE" : "SHORT_OVER_PAY";
        capitalFlowRepository.save(CapitalFlow.of(c.getContractId(),
            type, settleAmount.abs(), settleAmount.signum() >= 0 ? "短溢货款收取" : "短溢货款支付"));
        c.setShortOverPaymentSettled(true);
    }
}

关键决策：多租户可配置化

平台服务 100+ 客户，每家的合同模板、交收方式、资金规则、审批流都不同。若为每家写分支，代码必然膨胀。决策：一套引擎 + 每客户一套配置——用 Drools 规则 + 策略模式 + 配置化模板，把"差异"从代码移到配置。 

/**
 * 合同模板 / 交收方式 / 资金规则从租户配置读取，
 * 新增客户只加配置，不改代码 —— 100+ 客户差异化需求下代码零膨胀。
 */
@Service
public class ContractTemplateResolver {

    @Autowired private TenantConfigRepository tenantConfigRepository;
    @Autowired private KieContainer kieContainer; // Drools 规则引擎

    public ContractTemplate resolve(String tenantId, String tradeMode) {
        TenantConfig cfg = tenantConfigRepository.get(tenantId);
        // 1) 基础模板（审批节点、交收方式、必填项）来自配置
        ContractTemplate tpl = cfg.templateOf(tradeMode);
        // 2) 差异化规则（资金比例、开票时点、短溢阈值）走 Drools
        KieSession session = kieContainer.newKieSession(cfg.ruleSet());
        session.insert(tpl);
        session.fireAllRules();
        return tpl;
    }
}

踩坑与解决

坑 1 · 跨月长事务：合同履约从签约到完成可能跨月，早期用数据库本地事务把"付款+仓单过户+开票+状态更新"耦合在一个事务里，导致长事务锁表、外部系统超时无法回滚。
解：拆成 Saga + 状态机 + 事件溯源——每个履约节点是独立 Saga 步骤，状态迁移由状态机校验，领域事件落事件表；任一步失败按补偿动作回滚，跨月长流程不再依赖一个大事务。

坑 2 · 短溢货款对账：实际发货量与合同量差异（石油、煤炭按批次称重）导致货款与预付款对不平，早期靠人工台账核对，效率低且易错。
解：短溢货款单独建流水类型、单独结算动作，并把"短溢"口径写入资金流水的唯一事实；日终对账以流水为准自动比对，把对账从 9 人天压缩到 2 小时（详见"合同履约"页）。
